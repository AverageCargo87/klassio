```typescript
export async function POST(req: NextRequest) {
  const session = await auth()
  if (!session?.user?.id) {
    return new Response(
      JSON.stringify({ error: 'Войдите в систему' }),
      { status: 401, headers: { 'Content-Type': 'application/json' } },
    )
  }
  let prompt = ''; let lessonId = ''
  try {
    const body = (await req.json()) as { prompt?: unknown; lessonId?: unknown }
    prompt = typeof body.prompt === 'string' ? body.prompt.trim() : ''
    lessonId = typeof body.lessonId === 'string' ? body.lessonId.trim() : ''
  } catch {
    return new Response(JSON.stringify({ error: 'Неверный формат запроса' }), {
      status: 400, headers: { 'Content-Type': 'application/json' },
    })
  }
  if (!lessonId) {
    return new Response(JSON.stringify({ error: 'lessonId обязателен' }), {
      status: 400, headers: { 'Content-Type': 'application/json' },
    })
  }
  const lessonRow = await db
    .select({ id: lessons.id })
    .from(lessons)
    .where(and(eq(lessons.id, lessonId), eq(lessons.userId, session.user.id)))
    .limit(1)
  if (lessonRow.length === 0) {
    return new Response(JSON.stringify({ error: 'Этот урок не ваш' }), {
      status: 403, headers: { 'Content-Type': 'application/json' },
    })
  }
  // ... env guard + downstream call ...
}
```

    ```typescript
    export async function getSignedUrl(agentId: string, apiKey: string): Promise<string> {
      const url = new URL('https://api.elevenlabs.io/v1/convai/conversation/get-signed-url')
      url.searchParams.set('agent_id', agentId)
      const response = await fetch(url.toString(), {
        method: 'GET',
        headers: { 'xi-api-key': apiKey },
        cache: 'no-store',
      })
      if (!response.ok) {
        const text = await response.text().catch(() => '')
        throw new Error(`11labs get-signed-url failed: HTTP ${response.status} ${text}`)
      }
      const data = (await response.json()) as { signed_url: string }
      if (!data.signed_url) throw new Error('11labs response missing signed_url field')
      return data.signed_url
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → /api/voice/signed-url | Authenticated browser POST from VoicePanel (06-02) crosses into Node runtime; carries `lessonId` (UUID, user-controllable from request body but cross-checked against session). |
| /api/voice/signed-url → api.elevenlabs.io | Server-to-server outbound HTTPS GET with `xi-api-key` header. Outbound TLS to a trusted vendor; key MUST NEVER appear in any client-shipped artifact. |
| process.env → request handler | Env vars `ELEVENLABS_API_KEY` + `ELEVENLABS_AGENT_ID` cross from runtime env into handler closure on every request. They MUST NOT leak via error messages, logs returned to client, or build artifacts. |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-06-01-01 | Information disclosure | `ELEVENLABS_API_KEY` env var | mitigate | Server-only env (no `NEXT_PUBLIC_` prefix). Route reads from `process.env` inside handler closure only. Acceptance test grep `NEXT_PUBLIC_ELEVENLABS` must find nothing. 06-02 E2E will assert key absent from page source / network requests (VOI-01-T). |
| T-06-01-02 | Information disclosure | `ELEVENLABS_AGENT_ID` env var | mitigate | Same as T-06-01-01 — server-only, never exposed to client. Agent ID is also semi-secret because it identifies our specific 11labs agent. |
| T-06-01-03 | Spoofing / Elevation of privilege | `POST /api/voice/signed-url` (auth + ownership) | mitigate | NextAuth `auth()` guard runs FIRST (before body parse — order verified by grep test). Ownership check `WHERE lesson.id = ? AND lesson.userId = session.user.id` returns 403 for both "not yours" and "doesn't exist" — no oracle. |
| T-06-01-04 | Tampering | Input validation on `lessonId` | mitigate | Defensive `typeof body.lessonId === 'string' ? body.lessonId.trim() : ''` rejects non-strings, empty strings, and whitespace-only strings. Drizzle prepared statement prevents SQL injection. Malformed JSON yields 400 (not 500). |
| T-06-01-05 | Information disclosure | Upstream error propagation | mitigate | 11labs errors caught and replaced with generic `'Не удалось подключиться к голосовому сервису'` (502). Raw error logged to server console only via `console.error('[voice/signed-url] ...', err)`. Stack trace never crosses the trust boundary back to client. |
| T-06-01-06 | Denial of service ($) | Cost amplification by mass signed-URL generation | accept | 11labs Daily call limit 100 + Concurrent call limit per subscription (PHASE-6-SETUP § 7). Each `/api/voice/signed-url` POST = exactly one upstream call. Per-user rate limit can be added in Phase 6.5 if abused. Accept for Phase 6 — closed beta, single dev user. |
| T-06-01-07 | Spoofing (CSRF) | Cross-site POST to /api/voice/signed-url | mitigate | NextAuth session cookie is `SameSite=Lax` (configured in Phase 1) — cross-site POST cannot include cookie, route returns 401. No state-changing side effects on our DB anyway. |

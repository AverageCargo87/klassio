<!-- Resend interception strategy -->
The Resend provider in auth.ts uses our custom sendVerificationRequest which calls fetch('https://api.resend.com/emails', ...).
In E2E we intercept this via Playwright `page.route` BEFORE submitting the login form, capture the body, parse the URL out of the html or text, then navigate the page to that URL.

Key consideration: page.route only intercepts requests originating from the BROWSER. Our fetch happens server-side (Server Action → signIn → Resend provider → fetch). Playwright cannot intercept server-side fetch directly.

Solution: instead of intercepting the network call, we read the verificationToken row from the live DB right after submitting the form. The token + identifier are inserted by NextAuth into the verificationToken table BEFORE Resend is called. We construct the magic link URL from those columns and navigate.
This is what RESEARCH § Validation Architecture line 1147 alludes to with "mock Resend, intercept email" — for our purposes, reading the DB row is the cleanest server-side intercept.

// E2E test fixture — direct Supabase access for setup/cleanup before each test.
// Uses DATABASE_URL_DIRECT (port 5432) — single connection, no pooler quirks.
import 'dotenv/config'
import { drizzle } from 'drizzle-orm/postgres-js'
import postgres from 'postgres'
import { eq, like, or } from 'drizzle-orm'
import * as schema from '../../lib/db/schema'

const TEST_EMAIL_PREFIX = 'e2e-test+'

let _client: ReturnType<typeof postgres> | null = null
let _db: ReturnType<typeof drizzle> | null = null

function getDb() {
  if (!_db) {
    if (!process.env.DATABASE_URL_DIRECT) {
      throw new Error('DATABASE_URL_DIRECT required for E2E tests')
    }
    _client = postgres(process.env.DATABASE_URL_DIRECT, { prepare: false, max: 1 })
    _db = drizzle({ client: _client, schema })
  }
  return _db
}

export async function closeTestDb() {
  if (_client) {
    await _client.end()
    _client = null
    _db = null
  }
}

/**
 * Insert (or upsert) the seed admin user + at least one upcoming lesson + whitelist the email.
 * Idempotent.
 */
export async function seedTestUser(email: string): Promise<{ userId: string; lessonId: string }> {
  const db = getDb()
  const lower = email.toLowerCase()

  await db.insert(schema.allowedEmails)
    .values({ email: lower, notes: `e2e ${new Date().toISOString()}` })
    .onConflictDoNothing()

  const [user] = await db.insert(schema.users)
    .values({ email: lower, childName: 'E2E-ребёнок', childAge: 10 })
    .onConflictDoUpdate({
      target: schema.users.email,
      set: { childName: 'E2E-ребёнок', childAge: 10 },
    })
    .returning()

  // Always create a fresh upcoming lesson for predictable card render
  const oneHourLater = new Date(Date.now() + 60 * 60 * 1000)
  const [lesson] = await db.insert(schema.lessons)
    .values({
      userId: user.id,
      scheduledAt: oneHourLater,
      topic: 'E2E тестовый урок',
      durationMin: 45,
      status: 'scheduled',
    })
    .returning()

  return { userId: user.id, lessonId: lesson.id }
}

/**
 * Remove all e2e test artifacts. Run in test.afterAll().
 * Looks for emails matching the e2e-test+ prefix.
 */
export async function resetTestDb(): Promise<void> {
  const db = getDb()
  const testUsers = await db.select({ id: schema.users.id, email: schema.users.email })
    .from(schema.users)
    .where(like(schema.users.email, `${TEST_EMAIL_PREFIX}%`))

  for (const u of testUsers) {
    await db.delete(schema.lessons).where(eq(schema.lessons.userId, u.id))
    await db.delete(schema.accounts).where(eq(schema.accounts.userId, u.id))
    await db.delete(schema.users).where(eq(schema.users.id, u.id))
  }

  await db.delete(schema.allowedEmails).where(like(schema.allowedEmails.email, `${TEST_EMAIL_PREFIX}%`))
  await db.delete(schema.verificationTokens).where(like(schema.verificationTokens.identifier, `${TEST_EMAIL_PREFIX}%`))
}

export function makeTestEmail(suffix: string): string {
  // Unique-per-spec email so parallel test files (if ever enabled) don't collide
  return `${TEST_EMAIL_PREFIX}${suffix}-${Date.now()}@example.com`
}

/**
 * Read the magic link URL from verificationToken table for a given email.
 * NextAuth inserts the row right BEFORE Resend send. The URL is constructed as
 * <AUTH_URL or baseURL>/api/auth/callback/resend?token=...&email=...
 */
export async function readMagicLinkFor(email: string, baseURL: string): Promise<string | null> {
  const db = getDb()
  const lower = email.toLowerCase()
  const rows = await db.select({
    token: schema.verificationTokens.token,
    identifier: schema.verificationTokens.identifier,
  })
    .from(schema.verificationTokens)
    .where(eq(schema.verificationTokens.identifier, lower))
    .limit(1)

  if (rows.length === 0) return null
  const { token, identifier } = rows[0]
  return `${baseURL}/api/auth/callback/resend?token=${encodeURIComponent(token)}&email=${encodeURIComponent(identifier)}`
}

```typescript
// Phase 8 — periodic checkpoint timer per D-03 channel #4.
// Every 10 minutes of an active session, fires a callback so the caller
// (VoicePanel) can sendContextualUpdate a state summary to Nataly. This
// places fresh state snapshots at the tail of her LLM context window
// where attention is strongest (mitigates lost-in-the-middle degradation
// per RESEARCH § Context Behavior, § Risk 5).
//
// Pure utility — no React, no SDK. VoicePanel composes this with the
// formatter from lib/contextual-updates and its own state refs.

const CHECKPOINT_INTERVAL_MS = 10 * 60 * 1000 // 10 minutes

export interface PeriodicCheckpointArgs {
  /** Wall-clock minutes elapsed since startPeriodicCheckpoint was called: 10, 20, 30, ... */
  elapsedMinutes: number
}

/**
 * Schedule a periodic checkpoint callback every 10 minutes.
 *
 * @param onCheckpoint Called with elapsedMinutes after each interval tick. Caller
 *                     supplies the actual data + format (typically formatPeriodicCheckpoint).
 * @returns Cleanup function that clears the interval. Idempotent — calling it
 *          multiple times after the timer is already cleared is a no-op.
 */
export function startPeriodicCheckpoint(
  onCheckpoint: (args: PeriodicCheckpointArgs) => void,
): () => void {
  let count = 0
  let timerId: ReturnType<typeof setInterval> | null = setInterval(() => {
    count += 1
    onCheckpoint({ elapsedMinutes: count * 10 })
  }, CHECKPOINT_INTERVAL_MS)

  return () => {
    if (timerId !== null) {
      clearInterval(timerId)
      timerId = null
    }
  }
}
```

CRITICAL conventions:
- Use `ReturnType<typeof setInterval>` for the timer handle — works in both Node and browser typings.
- The cleanup closure captures `timerId` by reference and nulls it after clearing — makes the cleanup idempotent (Wave 0 test #3).
- No React imports — this is a pure utility callable from any context (including non-React tests).
- The callback receives a structured object (`{elapsedMinutes}`) not just a number — leaves room for future fields without breaking the API.
- Interval constant is 10 minutes = 600_000 ms = 10 * 60 * 1000.

```typescript
// ── Phase 8 D-03 channel #4: periodic checkpoint every ~10 minutes ─────
// Anchors fresh state at the tail of Nataly's context window. Runs only
// while the session is active — starts on 'connected', clears on disconnect.
//
// SAFETY: dep is conversation.STATUS (string primitive — 'disconnected' |
// 'connecting' | 'connected' | 'disconnecting'), NOT conversation (object).
// This is the ONLY useEffect in this file with conversation-derived deps;
// it is safe because string identity changes only on actual transitions,
// not on every SDK mode-change. Phase 6.5 cleanup-bug pattern preserved.
useEffect(() => {
  if (conversation.status !== 'connected') return
  const cleanup = startPeriodicCheckpoint(({ elapsedMinutes }) => {
    try {
      const totalTasks = trainerConfig?.tasks?.length ?? 0
      const solvedCount = solvedTaskIdsRef.current.size
      const mistakeCount = mistakesRef.current.length
      convoCmdRef.current.sendContextualUpdate(
        formatPeriodicCheckpoint({ elapsedMinutes, solvedCount, totalTasks, mistakeCount }),
      )
    } catch (err) {
      console.error('[voice-panel] periodic checkpoint failed:', err)
    }
  })
  return cleanup
}, [conversation.status, trainerConfig?.tasks?.length])
```

CRITICAL conventions:
- The dep array uses `conversation.status` (string primitive — see Phase 6.5 lesson in PATTERNS.md). DO NOT change this to `[conversation]` (object) — that re-creates the timer on every status/mode update and breaks correctness.
- `trainerConfig?.tasks?.length` is included as a dep so the timer restarts if the trainer config changes mid-session (extremely unlikely but type-safe).
- Read refs INSIDE the tick callback (not in the closure body) — refs are reference-stable so the latest values are always seen.
- Cleanup function returned by startPeriodicCheckpoint is itself returned from useEffect — React handles disconnect → cleanup → status flip cycle correctly.
- DO NOT put this above the existing useLessonBusEvent block from plan 08-04 — keeping all bus subscriptions before the timer useEffect is the established pattern (visually obvious + matches voice-panel.test.tsx assertions about call order).
- DO NOT add another useEffect with `conversation` (object) in deps. Static check from plan 08-01's voice-panel-subs.test.tsx WILL catch this and fail RED.
- Use `convoCmdRef.current.sendContextualUpdate` (the new ref from plan 08-04), NOT `conversation.sendContextualUpdate` directly — keeps the same latched-ref discipline.

**Step A — lib/trainer/config-loader.ts (new, server-only):**
```typescript
// 'server-only' pragma ensures this is never bundled client-side
import 'server-only'
import * as fs from 'node:fs/promises'
import * as path from 'node:path'
import { trainerConfigSchema, type TrainerConfig } from './config-schema'

export async function loadTrainerConfig(filename: string): Promise<TrainerConfig> {
  const filePath = path.join(process.cwd(), 'public', 'trainer-configs', filename)
  const raw = await fs.readFile(filePath, 'utf8')
  const json = JSON.parse(raw)
  return trainerConfigSchema.parse(json)
}
```

  ```typescript
  const containerRef = useRef<HTMLDivElement>(null)
  const hintOverrides = useRef<Map<string, number>>(new Map())
  const [, forceUpdate] = useReducer((x: number) => x + 1, 0)

  useLessonBusEvent('trainer:highlight', ({ elementId, durationMs = 3000 }) => {
    const el = containerRef.current?.querySelector<HTMLElement>(`[data-task-id="${elementId}"]`)
    if (!el) { console.warn(`[TrainerPanel] highlight: element '${elementId}' not found`); return }
    el.classList.add('ring-2', 'ring-yellow-400', 'trainer-highlight')
    setTimeout(() => el.classList.remove('ring-2', 'ring-yellow-400', 'trainer-highlight'), durationMs)
  })

  useLessonBusEvent('trainer:show_hint', ({ taskId, hintLevel }) => {
    if (hintLevel > 3) { console.warn(`[TrainerPanel] show_hint: hintLevel ${hintLevel} > 3, ignoring`); return }
    hintOverrides.current.set(taskId, hintLevel)
    forceUpdate()
  })

  useLessonBusEvent('trainer:goto_task', ({ taskId }) => {
    const el = containerRef.current?.querySelector<HTMLElement>(`[data-task-id="${taskId}"]`)
    if (!el) { console.warn(`[TrainerPanel] goto_task: task '${taskId}' not found`); return }
    el.scrollIntoView({ behavior: 'smooth', block: 'nearest' })
    const input = el.querySelector<HTMLElement>('input, button')
    input?.focus()
  })
  ```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| public/ filesystem → config-loader → RSC | JSON files read from disk; parsed before passing to client |
| RSC (page.tsx) → client boundary (LessonShell) | TrainerConfig passed as serializable prop (no functions, no Date objects) |
| browser eval → trainer:highlight command | Command payload elementId used in querySelector — no innerHTML or eval |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-07-02-01 | Tampering | loadTrainerConfig path traversal | mitigate | Filename from `lesson.htmlTrainerPath` (DB column). path.join with fixed prefix `public/trainer-configs/` — path.join normalizes and cannot escape the prefix for simple filenames (no `../` in well-formed DB values). Admin-only write path. Acceptable risk for v1. |

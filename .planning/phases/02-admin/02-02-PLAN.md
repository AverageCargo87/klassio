<!-- D-03 Week grouping spec: -->
- Upcoming = lessons where scheduledAt is within next 4 calendar weeks (Mon-Sun, RU week)
- Calendar week = Monday 00:00 to Sunday 23:59:59 (Russian/European convention, per A5)
- Week 0 = the calendar week containing `now` (i.e., from this Monday to this Sunday)
- Week 1 = next calendar week
- Weeks 2, 3 = show as date range "дд-дд месяц" (e.g. "12-18 мая", "19-25 мая")
- Only weeks with ≥1 lesson appear (no empty week headers)
- Horizon = end of week 3 (i.e., the Sunday ending the 4th calendar week from this week)
- Lessons beyond horizon: not shown in upcoming section

WeekBucket type to export:
  export type WeekBucket = {
    label: string          // "Эта неделя" | "Следующая неделя" | "12-18 мая" | etc.
    weekIndex: number      // 0, 1, 2, or 3
    lessons: Lesson[]
  }

groupByWeek(lessons: Lesson[], now: Date): WeekBucket[]
  - Filter to lessons where scheduledAt >= now (upcoming only) AND within 4-week horizon
  - Group by weekIndex (0..3)
  - Generate label per weekIndex:
      0 → "Эта неделя"
      1 → "Следующая неделя"
      2, 3 → date range using Intl.DateTimeFormat('ru-RU', { day: 'numeric', month: 'long' })
              format: "{startDay}–{endDay} {month}" (en-dash, same month)
              if start/end month differ: "{startDay} {startMonth}–{endDay} {endMonth}"
  - Return only buckets with lessons.length > 0, sorted by weekIndex ascending

<!-- D-04 Smart-relative date formatting spec: -->
formatSmartDate(scheduledAt: Date, now: Date): string
  - Uses Intl.DateTimeFormat('ru-RU') throughout — no external libraries
  - Comparison: strip time from both dates, compare date-only
  - TODAY (same calendar date): "Сегодня в HH:mm"
  - TOMORROW (next calendar date): "Завтра в HH:mm"
  - THIS WEEK (same calendar week but not today/tomorrow): "Пятница, HH:mm"
    - Weekday in Russian nominative: Понедельник, Вторник, Среда, Четверг, Пятница, Суббота, Воскресенье
    - Use Intl.DateTimeFormat('ru-RU', { weekday: 'long' }) with value charAt(0).toUpperCase() + rest
  - FURTHER (different week): "Пт 15 мая, HH:mm"
    - Use Intl.DateTimeFormat('ru-RU', { weekday: 'short', day: 'numeric', month: 'long' })
    - Format: "{short_weekday} {day} {month}, HH:mm"
    - Short weekday: Intl.DateTimeFormat('ru-RU', { weekday: 'short' }) — пн, вт, ср, чт, пт, сб, вс
  - Time HH:mm: Intl.DateTimeFormat('ru-RU', { hour: '2-digit', minute: '2-digit' }) → "16:00"

   ```typescript
   import type { InferSelectModel } from 'drizzle-orm'
   import type { lessons } from '@/lib/db/schema'

   type Lesson = InferSelectModel<typeof lessons>

   export type WeekBucket = {
     label: string
     weekIndex: number
     lessons: Lesson[]
   }

   /** Returns Monday 00:00:00 of the week containing `date` */
   function getMondayOf(date: Date): Date {
     const d = new Date(date)
     const day = d.getDay() // 0=Sun, 1=Mon, ..., 6=Sat
     const diff = (day === 0 ? -6 : 1 - day) // shift to Monday
     d.setHours(0, 0, 0, 0)
     d.setDate(d.getDate() + diff)
     return d
   }

   function getWeekLabel(weekIndex: number, monday: Date): string {
     if (weekIndex === 0) return 'Эта неделя'
     if (weekIndex === 1) return 'Следующая неделя'
     // Weeks 2 and 3: "дд–дд месяц"
     const sunday = new Date(monday)
     sunday.setDate(monday.getDate() + 6)
     const fmt = new Intl.DateTimeFormat('ru-RU', { day: 'numeric', month: 'long' })
     const startDay = monday.getDate()
     const endDay = sunday.getDate()
     const startMonth = monday.toLocaleString('ru-RU', { month: 'long' })
     const endMonth = sunday.toLocaleString('ru-RU', { month: 'long' })
     if (startMonth === endMonth) {
       return `${startDay}–${endDay} ${startMonth}`
     }
     return `${startDay} ${startMonth}–${endDay} ${endMonth}`
   }

   export function groupByWeek(lessons: Lesson[], now: Date): WeekBucket[] {
     const thisMonday = getMondayOf(now)
     // Horizon: end of Sunday of week 3
     const horizonSunday = new Date(thisMonday)
     horizonSunday.setDate(thisMonday.getDate() + 4 * 7 - 1) // +27 days = end of week 3
     horizonSunday.setHours(23, 59, 59, 999)

     const buckets: Map<number, Lesson[]> = new Map()

     for (const lesson of lessons) {
       if (lesson.scheduledAt < now) continue // past — skip
       if (lesson.scheduledAt > horizonSunday) continue // beyond horizon — skip
       const lessonMonday = getMondayOf(lesson.scheduledAt)
       const weekIndex = Math.round(
         (lessonMonday.getTime() - thisMonday.getTime()) / (7 * 24 * 60 * 60 * 1000)
       )
       if (weekIndex < 0 || weekIndex > 3) continue
       if (!buckets.has(weekIndex)) buckets.set(weekIndex, [])
       buckets.get(weekIndex)!.push(lesson)
     }

     const result: WeekBucket[] = []
     for (const [weekIndex, weekLessons] of [...buckets.entries()].sort(([a], [b]) => a - b)) {
       const monday = new Date(thisMonday)
       monday.setDate(thisMonday.getDate() + weekIndex * 7)
       result.push({
         label: getWeekLabel(weekIndex, monday),
         weekIndex,
         lessons: weekLessons.sort((a, b) => a.scheduledAt.getTime() - b.scheduledAt.getTime()),
       })
     }
     return result
   }
   ```

4. Create app/lessons/smart-date.ts:
   ```typescript
   function isSameDate(a: Date, b: Date): boolean {
     return a.getFullYear() === b.getFullYear() &&
       a.getMonth() === b.getMonth() &&
       a.getDate() === b.getDate()
   }

   function isSameCalendarWeek(a: Date, b: Date): boolean {
     // Same Mon-Sun calendar week
     function getMonday(d: Date): Date {
       const copy = new Date(d)
       const day = copy.getDay()
       copy.setHours(0, 0, 0, 0)
       copy.setDate(copy.getDate() + (day === 0 ? -6 : 1 - day))
       return copy
     }
     return getMonday(a).getTime() === getMonday(b).getTime()
   }

   function formatTime(d: Date): string {
     return d.toLocaleString('ru-RU', { hour: '2-digit', minute: '2-digit' })
   }

   function capitalize(s: string): string {
     return s.charAt(0).toUpperCase() + s.slice(1)
   }

   export function formatSmartDate(scheduledAt: Date, now: Date): string {
     const time = formatTime(scheduledAt)

     if (isSameDate(scheduledAt, now)) {
       return `Сегодня в ${time}`
     }

     const tomorrow = new Date(now)
     tomorrow.setDate(now.getDate() + 1)
     if (isSameDate(scheduledAt, tomorrow)) {
       return `Завтра в ${time}`
     }

     if (isSameCalendarWeek(scheduledAt, now)) {
       const weekday = capitalize(
         scheduledAt.toLocaleString('ru-RU', { weekday: 'long' })
       )
       return `${weekday}, ${time}`
     }

     // Different week: "Пт 15 мая, 16:00"
     const shortWeekday = capitalize(
       scheduledAt.toLocaleString('ru-RU', { weekday: 'short' })
     )
     const day = scheduledAt.getDate()
     const month = scheduledAt.toLocaleString('ru-RU', { month: 'long' })
     return `${shortWeekday} ${day} ${month}, ${time}`
   }
   ```

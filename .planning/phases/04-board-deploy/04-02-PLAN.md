```typescript
import { describe, it, expect, vi, beforeEach } from 'vitest'
import { POST } from '../route'
import { NextRequest } from 'next/server'

// Mock auth
vi.mock('@/auth', () => ({
  auth: vi.fn(),
}))

// Mock db
vi.mock('@/lib/db', () => ({
  db: { select: vi.fn() },
}))
vi.mock('@/lib/db/schema', () => ({
  lessons: { id: 'id', userId: 'user_id' },
}))
vi.mock('drizzle-orm', () => ({
  eq: vi.fn((a, b) => ({ field: a, value: b })),
  and: vi.fn((...args) => args),
}))

// Mock OpenAI
vi.mock('openai', () => {
  return {
    default: vi.fn().mockImplementation(() => ({
      chat: {
        completions: {
          stream: vi.fn().mockReturnValue({
            on: vi.fn().mockReturnThis(),
            finalChatCompletion: vi.fn().mockResolvedValue({
              choices: [{
                message: { tool_calls: [] },
                finish_reason: 'stop',
              }],
              usage: { prompt_tokens: 100, completion_tokens: 20 },
            }),
          }),
        },
      },
    })),
  }
})

import { auth } from '@/auth'
import { db } from '@/lib/db'

function makeRequest(body: unknown) {
  return new NextRequest('http://localhost/api/draw', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(body),
  })
}

describe('POST /api/draw', () => {
  beforeEach(() => {
    vi.mocked(auth).mockReset()
    vi.mocked(db.select).mockReset()
  })

  it('returns 401 when not authenticated', async () => {
    vi.mocked(auth).mockResolvedValue(null as never)
    const res = await POST(makeRequest({ prompt: 'test', lessonId: 'uuid-1' }))
    expect(res.status).toBe(401)
  })

  it('returns 400 when prompt is empty', async () => {
    vi.mocked(auth).mockResolvedValue({ user: { id: 'user-1' } } as never)
    const res = await POST(makeRequest({ prompt: '', lessonId: 'uuid-1' }))
    expect(res.status).toBe(400)
  })

  it('returns 400 when lessonId is missing', async () => {
    vi.mocked(auth).mockResolvedValue({ user: { id: 'user-1' } } as never)
    const res = await POST(makeRequest({ prompt: 'объясни' }))
    expect(res.status).toBe(400)
  })

  it('returns 403 when lesson does not belong to user', async () => {
    vi.mocked(auth).mockResolvedValue({ user: { id: 'user-1' } } as never)
    // db.select chain: .from().where().limit() returns empty array
    const chainMock = { from: vi.fn().mockReturnThis(), where: vi.fn().mockReturnThis(), limit: vi.fn().mockResolvedValue([]) }
    vi.mocked(db.select).mockReturnValue(chainMock as never)
    const res = await POST(makeRequest({ prompt: 'объясни', lessonId: 'uuid-bad' }))
    expect(res.status).toBe(403)
  })

  it('returns SSE stream (200) when auth + lessonId valid', async () => {
    vi.mocked(auth).mockResolvedValue({ user: { id: 'user-1' } } as never)
    const chainMock = { from: vi.fn().mockReturnThis(), where: vi.fn().mockReturnThis(), limit: vi.fn().mockResolvedValue([{ id: 'uuid-1' }]) }
    vi.mocked(db.select).mockReturnValue(chainMock as never)
    // Mock stream's on() to trigger done immediately
    const onMock = vi.fn().mockReturnThis()
    const finalMock = vi.fn().mockResolvedValue({
      choices: [{ message: { tool_calls: [] }, finish_reason: 'stop' }],
      usage: { prompt_tokens: 50, completion_tokens: 10 },
    })
    const { default: OpenAI } = await import('openai')
    vi.mocked(OpenAI).mockImplementation(() => ({
      chat: { completions: { stream: vi.fn().mockReturnValue({ on: onMock, finalChatCompletion: finalMock }) } }
    }) as never)

    const res = await POST(makeRequest({ prompt: 'объясни 5+3', lessonId: 'uuid-1' }))
    expect(res.status).toBe(200)
    expect(res.headers.get('Content-Type')).toContain('text/event-stream')
  })
})
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| HTTP client → POST /api/draw | User-supplied prompt + lessonId cross server trust boundary |
| /api/draw → OpenAI API | Server sends prompt to external LLM; response trusted as tool calls |
| /api/draw → Neon DB | Ownership check query; lessonId treated as untrusted until verified |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-04-02-01 | Elevation of Privilege | POST /api/draw — unauthenticated access | mitigate | auth() called first; 401 on null session before any body parsing |
| T-04-02-02 | Elevation of Privilege | /api/draw — IDOR: lessonId not owned by user | mitigate | Drizzle query WHERE id=$lessonId AND userId=$session.user.id; 403 if no row (D-08) |
| T-04-02-03 | Denial of Service | runaway agent loop consuming OpenAI credits | mitigate | MAX_AGENT_TURNS=30 hard cap; finishedExplicitly exits early on finish tool |
| T-04-02-04 | Information Disclosure | OpenAI errors in SSE error event | accept | Error message from OpenAI may contain partial prompt; acceptable for server logs; client gets generic error string, not stack trace |
| T-04-02-05 | Tampering | prompt injection via lessonId or prompt field | mitigate | lessonId is DB-verified UUID (ownership check); prompt is passed to OpenAI as user content (no server-side eval); model is constrained to tool calls by tool_choice:'required' |
</threat_model>
